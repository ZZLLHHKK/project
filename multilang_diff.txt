+"""自然語言日期解析器，將"明天"、"4月26號"等轉換為標準日期。"""
+
+from __future__ import annotations
+
+import re
+from datetime import datetime, timedelta
+from typing import Optional
+
+
+class DateParser:
+    """解析自然語言日期到標準化的年月日、時分秒。"""
+
+    def __init__(self):
+        self.now = datetime.now()
+        self.today = self.now.date()
+
+    def parse(self, text: str) -> Optional[dict]:
+        """
+        解析文本中的日期和時間。
+        
+        支援格式:
+        - "明天 22:00" → {year: 2026, month: 4, day: 26, hour: 22, minute: 0, second: 0}
+        - "後天 14:30" → {year: 2026, month: 4, day: 27, hour: 14, minute: 30, second: 0}
+        - "4月26號 22:00" → {year: 2026, month: 4, day: 26, hour: 22, minute: 0, second: 0}
+        - "下週一 10:00" → {year: 2026, month: 4, day: 29, hour: 10, minute: 0, second: 0}
+        - "22:00" → {hour: 22, minute: 0, second: 0} (沒有日期，表示今天)
+        - "每天 22:00" → {recurrence: "daily", hour: 22, minute: 0, second: 0}
+        - "每週一 10:00" → {recurrence: "weekly", weekday: 0, hour: 10, minute: 0, second: 0}
+        
+        Args:
+            text: 使用者輸入的日期時間文本
+            
+        Returns:
+            標準化的日期時間字典，或 None 若解析失敗
+        """
+        if not text:
+            return None
+
+        text = text.strip()
+        result = {}
+
+        # 1. 先檢查重複模式
+        recurrence = self._parse_recurrence(text)
+        if recurrence:
+            result.update(recurrence)
+
+        # 2. 提取時間 (HH:MM 或 HH:MM:SS)
+        time_match = self._parse_time(text)
+        if time_match:
+            result.update(time_match)
+        else:
+            # 如果沒有找到時間，這個日期字符串無效
+            if not recurrence:
+                return None
+
+        # 3. 提取日期 (如果有的話)
+        if not recurrence:  # 重複排程不需要日期
+            date_match = self._parse_date(text)
+            if date_match:
+                result.update(date_match)
+            # 如果沒有日期，預設為今天 (recurrence == None)
+
+        return result if result else None
+
+    def _parse_recurrence(self, text: str) -> Optional[dict]:
+        """檢查是否為重複排程。"""
+        # 每天
+        if "每天" in text or "每日" in text:
+            return {"recurrence": "daily"}
+
+        # 每週 + 星期幾
+        weekly_match = re.search(r"每週([一二三四五六日])", text)
+        if weekly_match:
+            day_name = weekly_match.group(1)
+            weekday = self._day_name_to_weekday(day_name)
+            return {"recurrence": "weekly", "weekday": weekday}
+
+        # 每月
+        if "每月" in text:
+            # 提取日期 (如"每月26號")
+            day_match = re.search(r"每月(\d{1,2})(?:號)?", text)
+            if day_match:
+                day = int(day_match.group(1))
+                if 1 <= day <= 31:
+                    return {"recurrence": "monthly", "day": day}
+
+        return None
+
+    def _parse_date(self, text: str) -> Optional[dict]:
+        """提取日期 (年月日)。"""
+        # 明天
+        if "明天" in text:
+            tomorrow = self.today + timedelta(days=1)
+            return {
+                "year": tomorrow.year,
+                "month": tomorrow.month,
+                "day": tomorrow.day,
+                "recurrence": "once",
+            }
+
+        # 後天
+        if "後天" in text:
+            day_after_tomorrow = self.today + timedelta(days=2)
+            return {
+                "year": day_after_tomorrow.year,
+                "month": day_after_tomorrow.month,
+                "day": day_after_tomorrow.day,
+                "recurrence": "once",
+            }
+
+        # 月份 + 日期 (例: "4月26號" 或 "4/26" 或 "四月26")
+        # 匹配 "4月26號", "04月26日", "4-26", "4/26" 等
+        date_patterns = [
+            r"(\d{1,2})月(\d{1,2})(?:號|日)?",  # "4月26號"
+            r"(\d{1,2})[/\-](\d{1,2})",  # "4/26" 或 "4-26"
+        ]
+
+        # 中文月份對照
+        month_map = {
+            "一": 1, "二": 2, "三": 3, "四": 4, "五": 5, "六": 6,
+            "七": 7, "八": 8, "九": 9, "十": 10, "十一": 11, "十二": 12,
+        }
+
+        # 檢查中文月份
+        for cn_month, num_month in month_map.items():
+            if cn_month + "月" in text:
+                day_match = re.search(rf"{cn_month}月(\d{{1,2}})(?:號|日)?", text)
+                if day_match:
+                    day = int(day_match.group(1))
+                    if 1 <= day <= 31:
+                        return {
+                            "year": self.now.year,
+                            "month": num_month,
+                            "day": day,
+                            "recurrence": "once",
+                        }
+
+        # 檢查數字月份
+        for pattern in date_patterns:
+            match = re.search(pattern, text)
+            if match:
+                month = int(match.group(1))
+                day = int(match.group(2))
+
+                # 驗證合法性
+                if 1 <= month <= 12 and 1 <= day <= 31:
+                    return {
+                        "year": self.now.year,
+                        "month": month,
+                        "day": day,
+                        "recurrence": "once",
+                    }
+
+        # 下週一、下週二... (下週 + 星期)
+        next_week_match = re.search(r"下週([一二三四五六日])", text)
+        if next_week_match:
+            day_name = next_week_match.group(1)
+            weekday = self._day_name_to_weekday(day_name)
+            # 計算下週的該星期幾
+            days_ahead = weekday - self.now.weekday()
+            if days_ahead <= 0:  # 下週的話，至少要 7 天
+                days_ahead += 7
+            target_date = self.today + timedelta(days=days_ahead)
+            return {
+                "year": target_date.year,
+                "month": target_date.month,
+                "day": target_date.day,
+                "recurrence": "once",
+            }
+
+        return None
+
+    def _parse_time(self, text: str) -> Optional[dict]:
+        """提取時間 (時分秒)。"""
+        # 匹配 HH:MM:SS, HH:MM, H:MM 等格式
+        time_patterns = [
+            r"(\d{1,2}):(\d{2}):(\d{2})",  # "22:00:30"
+            r"(\d{1,2}):(\d{2})",  # "22:00"
+            r"(\d{1,2})點(\d{2})分(\d{2})秒",  # "22點00分30秒"
+            r"(\d{1,2})點(\d{2})分",  # "22點00分"
+            r"(\d{1,2})點",  # "22點"
+        ]
+
+        for pattern in time_patterns:
+            match = re.search(pattern, text)
+            if match:
+                groups = match.groups()
+                hour = int(groups[0])
+                minute = int(groups[1]) if len(groups) > 1 else 0
+                second = int(groups[2]) if len(groups) > 2 else 0
+
+                # 驗證合法性
+                if 0 <= hour <= 23 and 0 <= minute <= 59 and 0 <= second <= 59:
+                    return {
+                        "hour": hour,
+                        "minute": minute,
+                        "second": second,
+                    }
+
+        return None
+
+    def _day_name_to_weekday(self, day_name: str) -> int:
+        """將中文星期名轉換為 weekday (0=Monday, 6=Sunday)。"""
+        day_map = {
+            "一": 0,
+            "二": 1,
+            "三": 2,
+            "四": 3,
+            "五": 4,
+            "六": 5,
+            "日": 6,
+        }
+        return day_map.get(day_name, 0)
+
+
+def parse_date(text: str) -> Optional[dict]:
+    """便利函數，快速解析日期。"""
+    parser = DateParser()
+    return parser.parse(text)

+_CHINESE_MIN_RE = re.compile(r"\d{1,2}\s*點\s*(\d{1,2})\s*分")

+def _detect_period(text: str) -> str:
+    for w in _PERIOD_NOON:
+        if w in text:

+    for w in _PERIOD_NIGHT:
+        if w in text:
+            return "night"
+    for w in _PERIOD_PM:
+        if w in text:

+    for w in _PERIOD_AM:
+        if w in text:

+def parse_time(text: str) -> Optional[tuple[int, int]]:
+    """Extract (hour, minute) in 24-hour format from text.
+
+    Returns None when time is ambiguous (e.g. '8點' with no period marker).
+    Callers should route ambiguous text to Gemini for clarification.
+    """
+    # HH:MM is always unambiguous
+    m = _HHMM_RE.search(text)
+    if m:
+        h, mi = int(m.group(1)), int(m.group(2))
+        if 0 <= h <= 23 and 0 <= mi <= 59:
+            return h, mi
+
+    hour_m = _CHINESE_HOUR_RE.search(text)
+    if not hour_m:
+        return None
+
+    hour = int(hour_m.group(1))
+
+    minute = 0
+    if _CHINESE_HALF_RE.search(text):
+        minute = 30
+    else:
+        min_m = _CHINESE_MIN_RE.search(text)
+        if min_m:
+            minute = int(min_m.group(1))
+
+    period = _detect_period(text)
+
+    if period == "noon":
+        hour = 12 if hour == 12 else 12
+    elif period in ("pm", "night"):
+        if hour != 12:
+            hour += 12
+    elif period == "am":
+        if hour == 12:
+            hour = 0
+    else:
+        # No period marker: only treat as unambiguous if hour >= 13 (clear 24-h)
+        if hour < 13:
+            return None
+
+    if 0 <= hour <= 23 and 0 <= minute <= 59:
+        return hour, minute
+    return None
+

+def parse_sched_add(parts: list[str]) -> Optional[Tuple[int, int, list[dict], str]]:
+    """Parse: /sched add HH:MM fan on|off | led <loc> on|off | temp <val>
+    Returns (hour, minute, actions, name) or None on parse failure.
+    """
+    if len(parts) < 4:
+        return None
+    time_str = parts[2]
+    try:
+        hh, mm = time_str.split(":")
+        hour, minute = int(hh), int(mm)
+        if not (0 <= hour <= 23 and 0 <= minute <= 59):
+            return None
+    except Exception:
+        return None
+
+    device = parts[3].lower()
+
+    if device == "fan":
+        if len(parts) < 5:
+            return None
+        state = parts[4].lower()
+        if state not in ("on", "off"):
+            return None
+        actions = [{"type": "FAN", "state": state}]
+        name = f"{hour:02d}:{minute:02d} 風扇{'開' if state == 'on' else '關'}"
+        return hour, minute, actions, name
+
+    if device == "led":
+        if len(parts) < 6:
+            return None
+        loc_map = {"living": "LIVING", "kitchen": "KITCHEN", "guest": "GUEST",
+                   "客廳": "LIVING", "廚房": "KITCHEN", "客房": "GUEST"}
+        loc = loc_map.get(parts[4].lower())
+        if loc is None:
+            return None
+        state = parts[5].lower()
+        if state not in ("on", "off"):
+            return None
+        loc_label = {"LIVING": "客廳", "KITCHEN": "廚房", "GUEST": "客房"}.get(loc, loc)
+        actions = [{"type": "LED", "location": loc, "state": state}]
+        name = f"{hour:02d}:{minute:02d} {loc_label}燈{'開' if state == 'on' else '關'}"
+        return hour, minute, actions, name
+
+    if device == "temp":
+        if len(parts) < 5:
+            return None
+        try:
+            val = int(parts[4])
+        except Exception:

+"""語言偵測工具 - 多語言支援新增
+策略：計算 ASCII 英文字母數 vs CJK 中文字符數。
+英文字母 > 中文字符 且 英文字母 >= 2 → 判定為英文 ('en')，否則中文 ('zh')
+輕量無依賴，不影響任何現有流程。
+"""
+
+from __future__ import annotations
+
+def detect_lang(text: str) -> str:
+    """偵測輸入文字的主要語言，返回 'en' 或 'zh' (預設)。"""
+    clean = (text or "").strip()
+    if not clean:
+        return "zh"
+
+    # 多語言支援新增：只要有任何一個 CJK 字符，優先判定為中文
+    cjk_chars = sum(1 for c in clean if '一' <= c <= '鿿' or '㐀' <= c <= '䶿')
+    if cjk_chars > 0:
+        return "zh"
+
+    # 完全沒有中文字符，且英文字母 >= 2 → 判定為英文
+    ascii_letters = sum(1 for c in clean if c.isalpha() and ord(c) < 128)
+    if ascii_letters >= 2:
+        return "en"
+    return "zh"
